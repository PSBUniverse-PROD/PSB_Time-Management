create view public.vw_wfk_workflow_setup as
select
  wf.wf_id as workflow_id,
  ws.wfs_id as workflow_stage_id,
  sp.stageparticipant_id,
  sp.orgrole_id as org_role_id,
  sp.approvaltype_id,
  wf.wf_name,
  wf.wf_description,
  wf.is_active as workflow_is_active,
  app.app_id,
  app.app_name,
  app.module_key,
  comp.comp_id,
  comp.comp_name,
  comp.short_name as company_short_name,
  dept.dept_id,
  dept.dept_name,
  dept.short_name as department_short_name,
  ws.stage_name,
  ws.stage_description,
  ws.stage_order,
  ws.is_active as stage_is_active,
  st.stagetype_id,
  st.stagetype_name,
  st.stagetype_description,
  org.name as org_role_name,
  org.description as org_role_description,
  org.is_active as org_role_is_active,
  at.approvaltype_name,
  at.approvaltype_description,
  sp.is_active as stageparticipant_is_active,
  sp.created_at as stageparticipant_created_at
from
  wfk_m_stageparticipant sp
  left join wfk_s_workflowstages ws on ws.wfs_id = sp.wfs_id
  left join wfk_s_workflow wf on wf.wf_id = ws.wf_id
  left join wfk_s_stagetype st on st.stagetype_id = ws.stagetype_id
  left join wfk_s_orgrole org on org.orgrole_id = sp.orgrole_id
  left join wfk_s_approvaltype at on at.approvaltype_id = sp.approvaltype_id
  left join psb_s_application app on app.app_id = wf.app_id
  left join psb_s_company comp on comp.comp_id = wf.comp_id
  left join psb_s_department dept on dept.dept_id = wf.dept_id;
